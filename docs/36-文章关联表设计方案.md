# 文章关联表设计方案

## 方案概述

通过创建专门的文章关联表（ArticleRelation），预先计算并存储文章之间的关联关系，大幅减少数据库查询次数和计算负载。

## 当前问题分析

### 现有方案
```typescript
// app/blog/[slug]/page.tsx
const allPosts = await getAllPosts() // 查询100条文章
const relatedPosts = allPosts
  .filter((p) => p.slug !== slug && (p.category === post.category || p.tags.some((tag) => post.tags.includes(tag))))
  .slice(0, 6) // 应用层过滤
```

### 问题
1. **查询100条文章**：每次访问文章详情页都要查询100条文章
2. **应用层过滤**：在JavaScript代码中过滤关联逻辑
3. **无法使用索引**：tags字段的contains查询无法使用索引
4. **重复计算**：每次访问都重新计算关联关系
5. **性能低下**：查询时间500ms+

## 解决方案：文章关联表

### 表结构设计

```prisma
// prisma/schema.prisma

model ArticleRelation {
  id                Int      @id @default(autoincrement())
  
  // 关联文章
  articleId         Int
  article           Article   @relation("SourceArticle", fields: [articleId], references: [id], onDelete: Cascade)
  
  // 被关联文章
  relatedArticleId  Int
  relatedArticle    Article   @relation("RelatedArticle", fields: [relatedArticleId], references: [id], onDelete: Cascade)
  
  // 关联类型
  type              RelationType // category（同分类）| tag（同标签）| manual（手动）
  
  // 关联权重（用于排序）
  weight            Int       @default(0)
  
  // 匹配的标签（如果是同标签关联）
  matchedTags       String?   @db.Text // JSON数组，存储匹配的标签
  
  // 时间戳
  createdAt         DateTime  @default(now())
  updatedAt         DateTime  @updatedAt
  
  // 唯一约束：避免重复关联
  @@unique([articleId, relatedArticleId, type])
  @@index([articleId])
  @@index([relatedArticleId])
  @@index([type])
  @@index([weight])
  @@map("article_relations")
}

// 关联类型枚举
enum RelationType {
  CATEGORY // 同分类
  TAG      // 同标签
  MANUAL   // 手动关联
}
```

### 查询优化效果对比

| 方案 | 查询方式 | 查询次数 | 响应时间 | 是否使用索引 |
|------|---------|---------|----------|------------|
| **当前方案** | 查询100条+应用层过滤 | 2次 | 500ms+ | ❌ |
| **关联表方案** | JOIN查询关联表 | 1次 | 10-20ms | ✅ |

**性能提升**：25-50倍

### 3. 创建关联关系生成脚本

```typescript
// scripts/generate-article-relations.ts

import { PrismaClient } from '@prisma/client'
import { parseTags } from '../lib/article-tags'

const prisma = new PrismaClient()

/**
 * 为单篇文章生成关联关系
 */
async function generateRelationsForArticle(articleId: number) {
  const article = await prisma.article.findUnique({
    where: { id: articleId },
    include: {
      category: true,
    },
  })

  if (!article) return

  // 删除旧关联
  await prisma.articleRelation.deleteMany({
    where: { articleId },
  })

  // 1. 同分类关联（权重最高）
  const categoryPosts = await prisma.article.findMany({
    where: {
      id: { not: articleId },
      categoryId: article.categoryId,
      published: true,
      publishedAt: { lte: new Date() },
    },
    include: { category: true },
    orderBy: { publishedAt: 'desc' },
    take: 6, // 只取前6篇
  })

  for (const related of categoryPosts) {
    await prisma.articleRelation.create({
      data: {
        articleId,
        relatedArticleId: related.id,
        type: 'CATEGORY',
        weight: 100, // 同分类权重最高
      },
    })
  }

  // 2. 同标签关联
  const articleTags = parseTags(article.tags)
  if (articleTags.length > 0) {
    // 查询有相同标签的文章（排除已关联的）
    const tagPosts = await prisma.article.findMany({
      where: {
        id: { 
          notIn: [articleId, ...categoryPosts.map(p => p.id)],
        },
        published: true,
        publishedAt: { lte: new Date() },
      },
      include: { category: true },
      orderBy: { publishedAt: 'desc' },
      take: 10, // 候选10篇
    })

    // 在应用层计算标签匹配度
    const tagMatches = tagPosts.map(post => {
      const postTags = parseTags(post.tags)
      const matched = articleTags.filter(tag => postTags.includes(tag))
      return {
        post,
        matchedCount: matched.length,
        matchedTags: matched,
      }
    })
    .filter(item => item.matchedCount > 0)
    .sort((a, b) => b.matchedCount - a.matchedCount)
    .slice(0, 3) // 取匹配度最高的3篇

    for (const match of tagMatches) {
      await prisma.articleRelation.create({
        data: {
          articleId,
          relatedArticleId: match.post.id,
          type: 'TAG',
          weight: match.matchedCount * 10, // 根据匹配标签数计算权重
          matchedTags: JSON.stringify(match.matchedTags),
        },
      })
    }
  }

  console.log(`Generated relations for article ${articleId}`)
}

/**
 * 为所有文章生成关联关系
 */
async function generateAllRelations() {
  console.log('Starting article relations generation...')

  const articles = await prisma.article.findMany({
    where: { published: true },
    select: { id: true },
  })

  for (const article of articles) {
    await generateRelationsForArticle(article.id)
  }

  console.log(`Completed! Generated relations for ${articles.length} articles`)
}

// 执行
generateAllRelations()
  .catch(console.error)
  .finally(() => prisma.$disconnect())
```

### 4. 更新 lib/blog.ts

```typescript
// lib/blog.ts

import { unstable_cache } from 'next/cache'
import { prisma } from './prisma'

/**
 * 获取相关文章（使用关联表）
 */
export async function getRelatedPosts(
  articleId: number,
  limit: number = 6
): Promise<BlogPost[]> {
  try {
    const getCachedRelatedPosts = unstable_cache(
      async () => {
        const relations = await prisma.articleRelation.findMany({
          where: {
            articleId,
            relatedArticle: {
              published: true,
              publishedAt: { lte: new Date() },
            },
          },
          include: {
            relatedArticle: {
              include: {
                category: true,
                author: true,
              },
            },
          },
          orderBy: {
            weight: 'desc', // 按权重排序
          },
          take: limit,
        })

        return relations.map(r => ({
          id: r.relatedArticle.id,
          slug: r.relatedArticle.slug,
          title: r.relatedArticle.title,
          description: r.relatedArticle.description || '',
          date: r.relatedArticle.publishedAt instanceof Date
            ? r.relatedArticle.publishedAt.toISOString()
            : r.relatedArticle.publishedAt || r.relatedArticle.createdAt.toISOString(),
          author: r.relatedArticle.author.name || r.relatedArticle.author.email || 'besttimeguide.com Team',
          category: r.relatedArticle.category.name,
          tags: parseTags(r.relatedArticle.tags),
          content: r.relatedArticle.content,
          readingTime: r.relatedArticle.readingTime || calculateReadingTime(r.relatedArticle.content),
          featured: r.relatedArticle.featured,
          coverImage: r.relatedArticle.coverImage,
        }))
      },
      [`related-posts-${articleId}`],
      {
        tags: [`related-posts`, `related-posts-${articleId}`],
        revalidate: 3600, // 1小时
      }
    )
    
    return await getCachedRelatedPosts()
  } catch (error) {
    console.error('Error fetching related posts:', error)
    return []
  }
}
```

### 6. 创建自动更新关联的API

```typescript
// app/api/articles/[id]/regenerate-relations/route.ts

import { NextRequest, NextResponse } from 'next/server'
import { auth } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

// 检查管理员权限
async function checkAdmin() {
  const session = await auth()
  if (!session?.user?.id) return null
  
  const userId = typeof session.user.id === 'string' 
    ? parseInt(session.user.id, 10) 
    : session.user.id
  
  const user = await prisma.user.findUnique({
    where: { id: userId },
    select: { isAdmin: true },
  })
  
  return user?.isAdmin ? userId : null
}

// POST - 重新生成单篇文章的关联关系
export async function POST(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const adminId = await checkAdmin()
  if (!adminId) {
    return NextResponse.json({ success: false, error: 'Unauthorized' }, { status: 401 })
  }

  const { id } = await params
  const articleId = parseInt(id, 10)

  try {
    // 导入生成函数
    const { generateRelationsForArticle } = await import('../../../../scripts/generate-article-relations')
    
    await generateRelationsForArticle(articleId)
    
    return NextResponse.json({ 
      success: true, 
      message: `Relations regenerated for article ${articleId}` 
    })
  } catch (error) {
    console.error('Error regenerating relations:', error)
    return NextResponse.json({ 
      success: false, 
      error: 'Internal server error' 
    }, { status: 500 })
  }
}
```
